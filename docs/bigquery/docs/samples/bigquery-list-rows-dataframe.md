    project = "bigquery-public-data"
    dataset_id = "samples"
    
    dataset_ref = bigquery.DatasetReference(project, dataset_id)
    table_ref = dataset_ref.table("shakespeare")
    table = client.get_table(table_ref)
    
    df = client.list_rows(table).to_dataframe()
